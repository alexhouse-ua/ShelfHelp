### New CI/CD Workflow

The primary deliverable for this story is the updated `.github/workflows/ci.yml` file. The developer agent must replace the existing content with the following verified configuration:

```yaml
name: CI

on:
  push:
    branches: ["main"]
  pull_request:
    branches: ["main"]
  workflow_dispatch:

jobs:
  # Job 1: Fast checks for code style and formatting
  lint-and-format:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      - name: Setup Deno
        uses: denoland/setup-deno@v2
        with:
          deno-version: v2.x
      # Cache Deno dependencies based on the lock file
      - name: Cache Deno dependencies
        uses: actions/cache@v4
        with:
          path: ~/.cache/deno
          key: ${{ runner.os }}-deno-${{ hashFiles('**/deno.lock') }}
          restore-keys: |
            ${{ runner.os }}-deno-
      - name: Run Deno lint
        run: deno lint
      - name: Run Deno format check
        run: deno fmt --check --ignore=docs/,README.md

  # Job 2: Fast checks for pure business logic
  unit-tests:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      - name: Setup Deno
        uses: denoland/setup-deno@v2
        with:
          deno-version: v2.x
      # Use the same dependency cache as the lint job
      - name: Cache Deno dependencies
        uses: actions/cache@v4
        with:
          path: ~/.cache/deno
          key: ${{ runner.os }}-deno-${{ hashFiles('**/deno.lock') }}
          restore-keys: |
            ${{ runner.os }}-deno-
      # Run only unit tests, which are fast and have no external dependencies
      - name: Run Unit Tests
        run: deno test -A --ignore=**/*.integration.test.ts

  # Job 3: Slower, more comprehensive tests that require a full database
  integration-tests:
    runs-on: ubuntu-latest # Using a standard runner
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      - name: Setup Deno
        uses: denoland/setup-deno@v2
        with:
          deno-version: v2.x
      # Use the same dependency cache
      - name: Cache Deno dependencies
        uses: actions/cache@v4
        with:
          path: ~/.cache/deno
          key: ${{ runner.os }}-deno-${{ hashFiles('**/deno.lock') }}
          restore-keys: |
            ${{ runner.os }}-deno-
      - name: Setup Supabase CLI
        uses: supabase/setup-cli@v1
        with:
          version: latest
      # Cache the Supabase Docker images to speed up 'supabase start'
      - name: Cache Supabase Docker images
        uses: actions/cache@v4
        with:
          path: /tmp/.docker-images-supabase.tar
          key: ${{ runner.os }}-supabase-docker-images
      # Load cached Docker images if they exist
      - name: Load Docker images
        run: |
          if [ -f /tmp/.docker-images-supabase.tar ]; then
            docker load --input /tmp/.docker-images-supabase.tar
          fi
      - name: Start Supabase local instance
        run: supabase start
      # Save Docker images to the cache path for the next run
      - name: Save Docker images
        run: |
          docker images --format '{{.Repository}}:{{.Tag}}' | grep 'supabase/' | xargs docker save --output /tmp/.docker-images-supabase.tar
      # Run only integration tests
      - name: Run Integration Tests
        run: deno test -A --ignore=**/*.unit.test.ts
        env:
          SUPABASE_URL: http://127.0.0.1:54321
          SUPABASE_SERVICE_ROLE_KEY: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZS1kZW1vIiwicm9sZSI6InNlcnZpY2Vfcm9sZSIsImV4cCI6MTk4MzgxMjk5Nn0.EGIM96RAZx35lJzdJsyH-qQwv8Hdp7fsn3W0YpN81IU
```

#### Unit Test Example (for Subtask 1.3)

```typescript
// Example: tests/scoring_logic.unit.test.ts
import { assertEquals } from "https://deno.land/std@0.208.0/assert/mod.ts";
import { calculateBookScore } from "../supabase/functions/_shared/scoring.ts";

Deno.test("calculateBookScore - returns higher score for recent additions", () => {
  const recentBook = {
    added_at: new Date().toISOString(),
    genre: "fiction",
    page_count: 300,
  };

  const oldBook = {
    added_at: new Date(Date.now() - 365 * 24 * 60 * 60 * 1000).toISOString(),
    genre: "fiction",
    page_count: 300,
  };

  const recentScore = calculateBookScore(recentBook);
  const oldScore = calculateBookScore(oldBook);

  assertEquals(recentScore > oldScore, true);
});
```

#### Integration Test Example (for Subtask 1.4)

```typescript
// Example: tests/book_addition.integration.test.ts
import { assertEquals } from "https://deno.land/std@0.208.0/assert/mod.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
import { cleanupTestData, seedTestData } from "./_test_utils.ts";

Deno.test("addBook - saves book with enriched metadata to database", async () => {
  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  );

  // Seed test user
  await seedTestData(supabase, {
    users: [{ id: "test-user-1", telegram_id: 12345 }],
  });

  // Test book addition
  const { data, error } = await supabase
    .from("books")
    .insert({
      user_id: "test-user-1",
      title: "Test Book",
      author: "Test Author",
      status: "to_read",
    })
    .select()
    .single();

  assertEquals(error, null);
  assertEquals(data?.title, "Test Book");

  // Cleanup
  await cleanupTestData(supabase);
});
```
